━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Para cada servidor:

1. Altere o Startup Command:
   
   Velocity:
   java -Xms128M -XX:MaxRAMPercentage=95.0 -jar velocity.jar
   
   Auth/Hub/Game:
   java -Xms1G -Xmx2G -jar paper.jar --nogui

2. Faça upload de TODOS os arquivos da pasta correspondente
   (mantenha a estrutura de pastas!)

3. Inicie o servidor

Guia completo: PTERODACTYL-GUIDE.md
